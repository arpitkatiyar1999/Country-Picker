Utility Functions

CountryPickerUtils provides powerful functions for phone number validation, formatting, and example generation using Google's libphonenumber library.

Import: import com.arpitkatiyarprojects.countrypicker.utils.CountryPickerUtils

isMobileNumberValid()

Validates a mobile number for a specific country using Google's libphonenumber library.

Signature

fun isMobileNumberValid(
    mobileNumber: String,
    countryCode: String
): Boolean

Parameters

ParameterTypeDescription
mobileNumberStringThe mobile number to validate (without country phone code)
countryCodeStringISO 3166-1 alpha-2 country code (e.g., "US", "IN", "GB")

Returns

Boolean - true if valid, false otherwise

Examples

// Valid numbers
val isValid1 = CountryPickerUtils.isMobileNumberValid("2025550123", "US")
// Returns: true

val isValid2 = CountryPickerUtils.isMobileNumberValid("7911123456", "GB")
// Returns: true

val isValid3 = CountryPickerUtils.isMobileNumberValid("9876543210", "IN")
// Returns: true

// Invalid numbers
val isValid4 = CountryPickerUtils.isMobileNumberValid("123", "US")
// Returns: false (too short)

val isValid5 = CountryPickerUtils.isMobileNumberValid("12345678901234567890", "US")
// Returns: false (too long)
Important: Do NOT include the country phone code in the mobile number. 
// ✗ Wrong
isMobileNumberValid("+1 202 555 0123", "US")

// ✓ Correct
isMobileNumberValid("2025550123", "US")

getExampleMobileNumber()

Retrieves an example mobile number for a given country, optionally formatted.

Signature

fun getExampleMobileNumber(
    countryCode: String,
    formatNumber: Boolean = false
): String

Parameters

ParameterTypeDefaultDescription
countryCodeString-ISO 3166-1 alpha-2 country code
formatNumberBooleanfalseIf true, returns formatted number; otherwise raw

Returns

String - Example number, or empty string if unavailable

Examples

// Raw format
val rawUS = CountryPickerUtils.getExampleMobileNumber("US", formatNumber = false)
// Returns: "2015550123"

val rawGB = CountryPickerUtils.getExampleMobileNumber("GB", formatNumber = false)
// Returns: "7400123456"

// Formatted
val formattedUS = CountryPickerUtils.getExampleMobileNumber("US", formatNumber = true)
// Returns: "201 555 0123"

val formattedGB = CountryPickerUtils.getExampleMobileNumber("GB", formatNumber = true)
// Returns: "7400 123456"

val formattedIN = CountryPickerUtils.getExampleMobileNumber("IN", formatNumber = true)
// Returns: "99876 54321"

getFormattedMobileNumber()

Formats a mobile number according to the country's specific formatting rules.

Requirement: This function should only be called AFTER validating the number with isMobileNumberValid(). It will return the original number unchanged if invalid.

Signature

fun getFormattedMobileNumber(
    mobileNumber: String,
    countryCode: String
): String

Parameters

ParameterTypeDescription
mobileNumberStringThe raw mobile number to format
countryCodeStringISO 3166-1 alpha-2 country code

Returns

String - Formatted number if valid, original number if invalid or on error

Examples

// US formatting
val formatted1 = CountryPickerUtils.getFormattedMobileNumber("2025550123", "US")
// Returns: "202 555 0123"

// UK formatting
val formatted2 = CountryPickerUtils.getFormattedMobileNumber("7911123456", "GB")
// Returns: "7911 123456"

// India formatting
val formatted3 = CountryPickerUtils.getFormattedMobileNumber("9876543210", "IN")
// Returns: "98765 43210"

// Invalid number - returns as-is
val formatted4 = CountryPickerUtils.getFormattedMobileNumber("123", "US")
// Returns: "123" (unchanged)

Complete Example

Combining all utility functions with phone number validation and formatting:

Basic Usage

@Composable
fun PhoneNumberForm() {
    // 1. Store phone number and selected country
    var phoneNumber by remember { mutableStateOf("") }
    var selectedCountry by remember { mutableStateOf(null) }

    Column {
        // 2. Phone input with country picker
        CountryPickerOutlinedTextField(
            mobileNumber = phoneNumber,
            onMobileNumberChange = { phoneNumber = it },
            onCountrySelected = { selectedCountry = it },
            label = { Text("Phone Number") }
        )

        // 3. Validate button
        Button(
            onClick = {
                selectedCountry?.let { country ->
                    // Validate the phone number
                    val isValid = CountryPickerUtils.isMobileNumberValid(
                        phoneNumber,
                        country.countryCode
                    )

                    if (isValid) {
                        // Get formatted number
                        val formatted = CountryPickerUtils.getFormattedMobileNumber(
                            phoneNumber,
                            country.countryCode
                        )
                        println("Valid: ${country.countryPhoneNumberCode} $formatted")
                    } else {
                        println("Invalid phone number")
                    }
                }
            }
        ) {
            Text("Validate")
        }
    }
}

With Real-Time Validation

@Composable
fun PhoneWithValidation() {
    var phoneNumber by remember { mutableStateOf("") }
    var selectedCountry by remember { mutableStateOf(null) }
    var isValid by remember { mutableStateOf(false) }

    // Validate whenever phone number or country changes
    LaunchedEffect(phoneNumber, selectedCountry) {
        isValid = selectedCountry?.let { country ->
            phoneNumber.isNotEmpty() &&
            CountryPickerUtils.isMobileNumberValid(phoneNumber, country.countryCode)
        } ?: false
    }

    CountryPickerOutlinedTextField(
        mobileNumber = phoneNumber,
        onMobileNumberChange = { phoneNumber = it },
        onCountrySelected = { selectedCountry = it },
        isError = phoneNumber.isNotEmpty() && !isValid,
        supportingText = {
            when {
                isValid -> Text("✓ Valid number", color = Color.Green)
                phoneNumber.isNotEmpty() -> Text("Invalid number", color = Color.Red)
                else -> Text("Enter phone number")
            }
        }
    )

    // Show formatted number when valid
    if (isValid && selectedCountry != null) {
        val formatted = CountryPickerUtils.getFormattedMobileNumber(
            phoneNumber,
            selectedCountry!!.countryCode
        )
        Text("Formatted: ${selectedCountry!!.countryPhoneNumberCode} $formatted")
    }
}
