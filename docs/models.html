Data Models

Complete reference for all data classes and enums used in Country-Picker library.

CountryDetails

Represents complete information about a country including code, phone code, name, and flag.

Declaration

@Immutable
data class CountryDetails(
    val countryCode: String,
    val countryPhoneNumberCode: String,
    val countryName: String,
    @DrawableRes val countryFlag: Int
)

Parameters

Parameter Type Description
countryCode String ISO 3166-1 alpha-2 country code (e.g., "US", "IN", "GB")
countryPhoneNumberCode String International dialing code with + prefix (e.g., "+1", "+91", "+44")
countryName String Full name of the country (e.g., "United States", "India", "United Kingdom")
countryFlag Int (@DrawableRes) Drawable resource ID of the country's flag image

Example Usage

var selectedCountry by remember { mutableStateOf(null) }

CountryPicker(
    onCountrySelected = { country ->
        selectedCountry = country
        println("Selected: ${country.countryName}")
        println("Code: ${country.countryCode}")
        println("Phone: ${country.countryPhoneNumberCode}")
    }
)

// Access country details
selectedCountry?.let { country ->
    Text("${country.countryName} (${country.countryPhoneNumberCode})")
    Image(painter = painterResource(country.countryFlag), contentDescription = null)
}

CountryListDisplayType

Enum defining how the country selection list is displayed to the user.

Declaration

enum class CountryListDisplayType {
    BottomSheet,
    Dialog
}

Values

Value Description
BottomSheet Displays the country list as a bottom sheet that slides up from the bottom of the screen. Recommended for mobile-first designs.
Dialog Displays the country list in a centered dialog box. Better for larger screens or when you want to focus user attention.

Example Usage

CountryPicker(
    countryListDisplayType = CountryListDisplayType.BottomSheet,
    onCountrySelected = { country -> /* ... */ }
)

// Or use Dialog
CountryPicker(
    countryListDisplayType = CountryListDisplayType.Dialog,
    onCountrySelected = { country -> /* ... */ }
)

CountryPickerColors

Defines all color customizations for the Country Picker component.

Declaration

@Immutable
data class CountryPickerColors(
    val dropDownIconColor: Color,
    val backIconColor: Color,
    val searchIconColor: Color,
    val cancelIconColor: Color,
    val searchCursorColor: Color,
    val selectedCountryContainerColor: Color,
    val countriesListContainerColor: Color,
    val selectedCountryDisabledContainerColor: Color,
    val dropDownDisabledIconColor: Color
)

Parameters

Parameter Type Description
dropDownIconColor Color Color of the dropdown arrow icon
backIconColor Color Color of the back/close icon in dialog/bottom sheet
searchIconColor Color Color of the search icon in the search bar
cancelIconColor Color Color of the cancel/clear icon in search bar
searchCursorColor Color Color of the text cursor in search field
selectedCountryContainerColor Color Background color of selected country container
countriesListContainerColor Color Background color of the countries list dialog/bottom sheet
selectedCountryDisabledContainerColor Color Background color when country picker is disabled
dropDownDisabledIconColor Color Color of dropdown icon when disabled

Example Usage

CountryPicker(
    colors = CountryPickerColors(
        dropDownIconColor = Color.Blue,
        backIconColor = Color.Black,
        searchIconColor = Color.Gray,
        cancelIconColor = Color.Red,
        searchCursorColor = Color.Blue,
        selectedCountryContainerColor = Color.Transparent,
        countriesListContainerColor = Color.White,
        selectedCountryDisabledContainerColor = Color.LightGray,
        dropDownDisabledIconColor = Color.Gray
    ),
    onCountrySelected = { /* ... */ }
)

CountryPickerDialogTextStyles

Customizes text appearance in the country picker dialog/bottom sheet.

Declaration

@Immutable
data class CountryPickerDialogTextStyles(
    val countryPhoneCodeTextStyle: TextStyle? = null,
    val countryNameTextStyle: TextStyle? = null,
    val countryCodeTextStyle: TextStyle? = null,
    val searchBarHintTextStyle: TextStyle? = null,
    val titleTextStyle: TextStyle? = null,
    val noSearchedCountryAvailableTextStyle: TextStyle? = null
)

Parameters

Parameter Type Default Description
countryPhoneCodeTextStyle TextStyle? null Text style for country phone code (e.g., "+1")
countryNameTextStyle TextStyle? null Text style for country name (e.g., "United States")
countryCodeTextStyle TextStyle? null Text style for country code (e.g., "US")
searchBarHintTextStyle TextStyle? null Text style for search bar placeholder text
titleTextStyle TextStyle? null Text style for dialog title (e.g., "Select a Country")
noSearchedCountryAvailableTextStyle TextStyle? null Text style for "no results" message

Example Usage

CountryPicker(
    dialogTextStyles = CountryPickerDialogTextStyles(
        countryPhoneCodeTextStyle = MaterialTheme.typography.bodyMedium.copy(
            fontWeight = FontWeight.Bold
        ),
        countryNameTextStyle = MaterialTheme.typography.bodyLarge,
        titleTextStyle = MaterialTheme.typography.headlineSmall.copy(
            color = Color.Blue
        ),
        searchBarHintTextStyle = MaterialTheme.typography.bodySmall.copy(
            color = Color.Gray
        )
    ),
    onCountrySelected = { /* ... */ }
)

BorderThickness

Defines border thickness for focused and unfocused states in CountryPickerOutlinedTextField.

Declaration

@Immutable
data class BorderThickness(
    val focusedBorderThickness: Dp = 2.dp,
    val unfocusedBorderThickness: Dp = 1.dp
)

Parameters

Parameter Type Default Description
focusedBorderThickness Dp 2.dp Border thickness when the text field is focused
unfocusedBorderThickness Dp 1.dp Border thickness when the text field is not focused

Example Usage

CountryPickerOutlinedTextField(
    mobileNumber = phoneNumber,
    onMobileNumberChange = { phoneNumber = it },
    onCountrySelected = { /* ... */ },
    borderThickness = BorderThickness(
        focusedBorderThickness = 3.dp,
        unfocusedBorderThickness = 1.5.dp
    )
)
