CountryPicker

A standalone composable that displays the country picker without any text field. Perfect for forms where you need country selection separate from input fields.

Function Signature

@Composable
fun CountryPicker(
    modifier: Modifier = Modifier,
    defaultPaddingValues: PaddingValues = PaddingValues(4.dp, 6.dp),
    selectedCountryDisplayProperties: SelectedCountryDisplayProperties = SelectedCountryDisplayProperties(),
    countriesListDialogDisplayProperties: CountriesListDialogDisplayProperties = CountriesListDialogDisplayProperties(),
    defaultCountryCode: String? = null,
    countriesList: List? = null,
    countryListDisplayType: CountryListDisplayType = CountryListDisplayType.Dialog,
    countryPickerColors: CountryPickerColors = CountryPickerDefault.colors(),
    isPickerEnabled: Boolean = true,
    onCountrySelected: (country: CountryDetails) -> Unit
)

Parameter Type Default Description
modifier Modifier Modifier Modifier to be applied to the component
defaultPaddingValues PaddingValues PaddingValues(4.dp, 6.dp) Internal padding between container and content
selectedCountryDisplayProperties SelectedCountryDisplayProperties SelectedCountryDisplayProperties() Display properties for the selected country (flag, text styles, spacing)
countriesListDialogDisplayProperties CountriesListDialogDisplayProperties CountriesListDialogDisplayProperties() Display properties for the country selection dialog/bottomsheet

Basic Usage

@Composable
fun BasicCountryPickerExample() {
    var selectedCountry by remember { mutableStateOf(null) }

    CountryPicker(
        onCountrySelected = { country ->
            selectedCountry = country
        }
    )
}

Customized Example

@Composable
fun CustomizedCountryPicker() {
    var selectedCountry by remember { mutableStateOf(null) }

    CountryPicker(
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
            .border(1.dp, Color.Gray, RoundedCornerShape(8.dp))
            .padding(12.dp),
        defaultCountryCode = "US",
        selectedCountryDisplayProperties = SelectedCountryDisplayProperties(
            properties = SelectedCountryProperties(
                showCountryFlag = true,
                showCountryPhoneCode = true,
                showCountryName = true,  // Show country name
                showCountryCode = false,
                showDropDownIcon = true
            ),
            flagDimensions = FlagDimensions(width = 32.dp, height = 22.dp),
            flagShape = RoundedCornerShape(4.dp),
            textStyles = SelectedCountryTextStyles(
                countryPhoneCodeTextStyle = TextStyle(
                    fontWeight = FontWeight.Bold,
                    fontSize = 16.sp
                ),
                countryNameTextStyle = TextStyle(
                    fontSize = 14.sp
                )
            )
        ),
        countryListDisplayType = CountryListDisplayType.BottomSheet,
        onCountrySelected = { country ->
            selectedCountry = country
        }
    )
}

CountryPickerOutlinedTextField

A comprehensive composable that combines the country picker with a Material 3 OutlinedTextField for phone number input. Includes built-in validation and formatting capabilities.

fun CountryPickerOutlinedTextField(

    modifier: Modifier = Modifier,
    defaultPaddingValues: PaddingValues = PaddingValues(4.dp, 6.dp),
    selectedCountryDisplayProperties: SelectedCountryDisplayProperties = SelectedCountryDisplayProperties(),
    countriesListDialogDisplayProperties: CountriesListDialogDisplayProperties = CountriesListDialogDisplayProperties(),
    defaultCountryCode: String? = null,
    countriesList: List? = null,
    enabled: Boolean = true,

    trailingIcon: @Composable (() -> Unit)? = null,
    prefix: @Composable (() -> Unit)? = null,
    suffix: @Composable (() -> Unit)? = null,
    supportingText: @Composable (() -> Unit)? = null,

Note: There's also a TextFieldValue version of this component for advanced text selection and cursor control. See API Reference for details.

Key Parameters (Additional to CountryPicker)

Parameter Type Description

Basic Usage

@Composable
fun BasicPhoneInput() {
    var phoneNumber by remember { mutableStateOf("") }
    var selectedCountry by remember { mutableStateOf(null) }

    CountryPickerOutlinedTextField(
        mobileNumber = phoneNumber,
        onMobileNumberChange = { phoneNumber = it },
        onCountrySelected = { selectedCountry = it },
        label = { Text("Phone Number") }
    )
}

With Validation

@Composable
fun ValidatedPhoneInput() {
    var phoneNumber by remember { mutableStateOf("") }
    var selectedCountry by remember { mutableStateOf(null) }
    var isError by remember { mutableStateOf(false) }

    // Validate on country or number change
    LaunchedEffect(phoneNumber, selectedCountry) {
        if (phoneNumber.isNotEmpty() && selectedCountry != null) {
            isError = !CountryPickerUtils.isMobileNumberValid(
                phoneNumber,
                selectedCountry!!.countryCode
            )
        }
    }

    CountryPickerOutlinedTextField(
        mobileNumber = phoneNumber,
        onMobileNumberChange = { phoneNumber = it },
        onCountrySelected = { selectedCountry = it },
        label = { Text("Phone Number") },
        placeholder = {
            selectedCountry?.let { country ->
                Text(
                    CountryPickerUtils.getExampleMobileNumber(
                        country.countryCode,
                        formatNumber = true
                    )
                )
            }
        },
        isError = isError,
        supportingText = {
            Text(
                text = when {
                    isError -> "Invalid phone number"
                    phoneNumber.isNotEmpty() -> "Valid phone number"
                    else -> "Enter your phone number"
                },
                color = when {
                    isError -> MaterialTheme.colorScheme.error
                    phoneNumber.isNotEmpty() -> Color(0xFF10B981)
                    else -> MaterialTheme.colorScheme.onSurfaceVariant
                }
            )
        }
    )
}

With Auto-Formatting

@Composable
fun FormattedPhoneInput() {
    var phoneNumber by remember { mutableStateOf(TextFieldValue("")) }
    var selectedCountry by remember { mutableStateOf(null) }

    CountryPickerOutlinedTextField(
        mobileNumber = phoneNumber,
        onMobileNumberChange = { newValue ->
            selectedCountry?.let { country ->
                // Format as user types
                val formatted = CountryPickerUtils.getFormattedMobileNumber(
                    newValue.text,
                    country.countryCode
                )
                phoneNumber = newValue.copy(
                    text = formatted,
                    selection = TextRange(formatted.length)
                )
            } ?: run {
                phoneNumber = newValue
            }
        },
        onCountrySelected = { selectedCountry = it },
        label = { Text("Phone Number") }
    )
}

Common Patterns

Disabled State

CountryPicker(
    isPickerEnabled = false,
    onCountrySelected = { /* ... */ }
)

// For TextField
CountryPickerOutlinedTextField(
    enabled = false,  // Disables both picker and text input
    // OR
    isPickerEnabled = false,  // Only disables the country picker
    // ...
)

Read-Only Phone Display

CountryPickerOutlinedTextField(
    mobileNumber = savedPhoneNumber,
    onMobileNumberChange = { /* no-op */ },
    onCountrySelected = { /* no-op */ },
    readOnly = true,
    enabled = false
)

Conditional Country List

val allowedCountries = remember {
    if (userType == UserType.DOMESTIC) {
        listOf("US")
    } else {
        null  // All countries
    }
}

CountryPicker(
    countriesList = allowedCountries,
    onCountrySelected = { /* ... */ }
)

With Callback on Done

CountryPickerOutlinedTextField(
    mobileNumber = phoneNumber,
    onMobileNumberChange = { phoneNumber = it },
    onCountrySelected = { selectedCountry = it },
    onDone = {
        // Handle form submission
        submitForm()
    }
)

Best Practices

✅ Always Provide Feedback

Use supportingText to show validation status, example numbers, or helpful hints.

✅ Validate Before Submission

Use CountryPickerUtils.isMobileNumberValid() before submitting forms.

✅ Show Example Numbers

Display country-specific example numbers in placeholders for better UX.

✅ Handle Empty States

Don't validate when the field is empty - only when user has entered input.

⚠️ Don't Include Country Code in Validation

Pass only the national number to validation functions, not the full international format. 

// ✗ Wrong
isMobileNumberValid("+1234567890", "US")

// ✓ Correct
isMobileNumberValid("234567890", "US")
